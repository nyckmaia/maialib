# Pitch Spelling Foundation (SP1) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace maiacore's hand-written pitch tables with arithmetic built on one pitch-string parser, supporting every `bb`/`b`/natural/`#`/`x` spelling in octaves -1..11.

**Architecture:** `Helper::splitPitch` becomes the only pitch-string parser; `Helper::pitch2midiNote` computes `12·(octave+1) + stepSemitone + alter`; `Note::getEnharmonicPitch` computes candidate spellings arithmetically with a range fallback; `Interval::whiteKeyDistance` computes diatonic indexes. Legacy behavior is pinned first by characterization tests generated from the old switch tables.

**Tech Stack:** C++17 (clang++ 18, CMake, GNU Make), GoogleTest 1.14, pybind11 2.13.6, Python unittest, Git Bash on Windows.

**Spec:** `docs/superpowers/specs/2026-09-15-pitch-spelling-design.md`

Write this file to `$TEMP/sp1_generate_pitch_spelling_data.py` (outside the repository):

```python
"""One-off SP1 extraction script. Run from the repository root before Task 2.

Reads the legacy switch tables of Helper::pitch2midiNote and Note::getEnharmonicPitch and
writes tests-cpp/src/pitch-spelling-legacy-data.h. It also embeds an independent Python model
of SP1 spec sections 6.2 and 6.4 for the full supported range.
"""

import pathlib
import re
import sys

ROOT = pathlib.Path.cwd()
SRC = ROOT / "maiacore" / "src" / "maiacore"
OUT = ROOT / "tests-cpp" / "src" / "pitch-spelling-legacy-data.h"

STEPS = "CDEFGAB"
SEMITONES = [0, 2, 4, 5, 7, 9, 11]
SYMBOL = {-2: "bb", -1: "b", 0: "", 1: "#", 2: "x"}
ALTER = {symbol: alter for alter, symbol in SYMBOL.items()}
PITCH_RE = re.compile(r"^([A-G])(bb|b|#|x)?(-?\d+)$")
MIN_OCTAVE, MAX_OCTAVE = -1, 11


def parse(pitch):
    match = PITCH_RE.match(pitch)
    if match is None:
        sys.exit(f"unparseable pitch: {pitch}")
    return STEPS.index(match.group(1)), ALTER[match.group(2) or ""], int(match.group(3))


def midi_number(pitch):
    step, alter, octave = parse(pitch)
    return 12 * (octave + 1) + SEMITONES[step] + alter


def spell(midi, alter):
    base = midi - alter
    pitch_class = base % 12
    if pitch_class not in SEMITONES:
        return ""
    octave = (base - pitch_class) // 12 - 1
    if not MIN_OCTAVE <= octave <= MAX_OCTAVE:
        return ""
    return f"{STEPS[SEMITONES.index(pitch_class)]}{SYMBOL[alter]}{octave}"


def enharmonics(pitch):
    """Independent model of spec section 6.4: rule plus range fallback."""
    _, own, _ = parse(pitch)
    midi = midi_number(pitch)
    other = {alter: spell(midi, alter) for alter in (-2, -1, 0, 1, 2) if alter != own}

    def first(alters):
        return next((other[a] for a in alters if other.get(a)), "")

    def get(alter):
        return other.get(alter, "")

    if midi % 12 in SEMITONES:  # white key
        flat_side, sharp_side, natural = first((-1, -2)), first((1, 2)), get(0)
        if own == 0:
            default, alternative = flat_side, sharp_side
        elif own < 0:
            default, alternative = natural, sharp_side
        else:
            default, alternative = natural, flat_side
    else:  # black key
        double = first((-2, 2))
        default, alternative = {
            1: (get(-1), double),
            -1: (get(1), double),
            2: (get(1), get(-1)),
            -2: (get(-1), get(1)),
        }[own]
    default = default or pitch
    alternative = alternative or default
    return default, alternative


def function_body(path, start_marker, end_marker):
    text = path.read_text(encoding="utf-8")
    start = text.index(start_marker)
    return text[start : text.index(end_marker, start)]


def legacy_midi_table():
    body = function_body(SRC / "helper.cpp", "int Helper::pitch2midiNote", "default:")
    table, pending = {}, []
    for line in body.splitlines():
        pending += re.findall(r'case hash\("([^"]+)"\):', line)
        match = re.search(r"return (?:MUSIC_XML::MIDI::NUMBER::MIDI_)?(\d+);", line)
        if match:
            for pitch in pending:
                table[pitch] = int(match.group(1))
            pending = []
    table["Db10"] = 133  # the legacy switch returned 132 (bug fixed in SP1)
    return table


def legacy_enharmonic_table():
    body = function_body(
        SRC / "note.cpp",
        "std::string Note::getEnharmonicPitch",
        "std::vector<std::string> Note::getEnharmonicPitches",
    )
    ternary = re.compile(
        r'enharmonicPitch = \(alternativeEnhamonicPitch\) \? "([^"]+)" : "([^"]+)";'
    )
    single = re.compile(r'enharmonicPitch = "([^"]+)";')
    table, pending = {}, []
    for line in body.splitlines():
        pending += re.findall(r'case hash\("([^"]+)"\):', line)
        if match := ternary.search(line):
            for pitch in pending:
                table[pitch] = (match.group(2), match.group(1))
            pending = []
        elif match := single.search(line):
            for pitch in pending:
                table[pitch] = (match.group(1), match.group(1))
            pending = []
    return table


def midi_block(name, rows, comment):
    lines = [f"// {comment}", f"inline constexpr std::array<PitchMidiEntry, {len(rows)}> {name} = {{{{"]
    lines += [f'    {{"{pitch}", {midi}}},' for pitch, midi in rows]
    return lines + ["}};", ""]


def enharmonic_block(name, rows, comment):
    lines = [
        f"// {comment}",
        f"inline constexpr std::array<PitchEnharmonicEntry, {len(rows)}> {name} = {{{{",
    ]
    lines += [f'    {{"{p}", "{d}", "{a}"}},' for p, d, a in rows]
    return lines + ["}};", ""]


def main():
    midi_table = legacy_midi_table()
    for pitch, value in midi_table.items():
        if midi_number(pitch) != value:
            sys.exit(f"legacy MIDI mismatch: {pitch} -> {value}")

    legacy_enharmonics = {
        pitch: value for pitch, value in legacy_enharmonic_table().items() if pitch in midi_table
    }
    for pitch, value in legacy_enharmonics.items():
        if enharmonics(pitch) != value:
            sys.exit(f"legacy enharmonic mismatch: {pitch} {value} != {enharmonics(pitch)}")

    full_range = [
        f"{STEPS[step]}{SYMBOL[alter]}{octave}"
        for octave in range(MIN_OCTAVE, MAX_OCTAVE + 1)
        for step in range(7)
        for alter in (-2, -1, 0, 1, 2)
        if 12 * (octave + 1) + SEMITONES[step] + alter >= 0
    ]

    def by_midi(pitch):
        return midi_number(pitch), pitch

    lines = [
        "// Generated once by the SP1 extraction script (docs/superpowers/plans/",
        "// 2026-09-15-pitch-spelling.md, Task 1) from the legacy switch tables of",
        "// Helper::pitch2midiNote and Note::getEnharmonicPitch. Do not edit by hand.",
        "#pragma once",
        "",
        "#include <array>",
        "",
        "struct PitchMidiEntry {",
        "    const char* pitch;",
        "    int midiNumber;",
        "};",
        "",
        "struct PitchEnharmonicEntry {",
        "    const char* pitch;",
        "    const char* defaultPitch;",
        "    const char* alternativePitch;",
        "};",
        "",
    ]
    lines += midi_block(
        "kLegacyMidiTable",
        [(p, midi_table[p]) for p in sorted(midi_table, key=by_midi)],
        'Legacy Helper::pitch2midiNote switch. "Db10" is 133: the legacy switch returned 132.',
    )
    lines += enharmonic_block(
        "kLegacyEnharmonicTable",
        [(p, *legacy_enharmonics[p]) for p in sorted(legacy_enharmonics, key=by_midi)],
        "Legacy Note::getEnharmonicPitch switch entries accepted by the legacy Note constructor.",
    )
    lines += midi_block(
        "kFullRangeMidiTable",
        [(p, midi_number(p)) for p in sorted(full_range, key=by_midi)],
        "All spellings in octaves -1..11 with MIDI >= 0 (independent model of spec 6.2).",
    )
    lines += enharmonic_block(
        "kFullRangeEnharmonicTable",
        [(p, *enharmonics(p)) for p in sorted(full_range, key=by_midi)],
        "Enharmonic rule and range fallback for every supported spelling (spec 6.4).",
    )
    OUT.write_text("\n".join(lines), encoding="utf-8", newline="\n")
    print(
        f"legacy MIDI: {len(midi_table)}; legacy enharmonic: {len(legacy_enharmonics)}; "
        f"full range: {len(full_range)}"
    )


if __name__ == "__main__":
    main()
```

void Helper::splitPitch(const std::string& pitch, std::string& pitchClass, std::string& pitchStep,
                        int& octave, float& alterValue, std::string& alterSymbol) {
    // Rest case: This is necessary to prevent: empty pitchClass + alterSymbol
    if (pitch.empty() || (pitch.find(MUSIC_XML::PITCH::REST) != std::string::npos)) {
        pitchClass = "rest";
        pitchStep = "rest";
        octave = 0;
        alterValue = 0;
        alterSymbol = "";
        return;
    }

    // ===== STEP ===== //
    const std::string step = pitch.substr(0, 1);
    const bool isValidStep = std::find(c_C_diatonicScale.begin(), c_C_diatonicScale.end(),
                                       step) != c_C_diatonicScale.end();
    if (!isValidStep) {
        LOG_ERROR("Unknown diatonic pitch: " + step);
    }

    // ===== OCTAVE ===== //
    // Trailing digits, optionally preceded by a single '-' (e.g. "C-1"). Default octave: 4
    size_t octaveStart = pitch.size();
    while (octaveStart > 1 && std::isdigit(static_cast<unsigned char>(pitch[octaveStart - 1]))) {
        octaveStart--;
    }
    const bool hasOctave = octaveStart < pitch.size();
    if (hasOctave && octaveStart > 1 && pitch[octaveStart - 1] == '-') {
        octaveStart--;
    }

    int parsedOctave = 4;
    if (hasOctave) {
        const std::string octaveToken = pitch.substr(octaveStart);
        parsedOctave = (octaveToken.size() <= 3) ? std::stoi(octaveToken) : c_maxPitchOctave + 1;
        if (parsedOctave < c_minPitchOctave || parsedOctave > c_maxPitchOctave) {
            LOG_ERROR("Invalid octave value: " + octaveToken);
        }
    }

    // ===== ACCIDENTAL ===== //
    const std::string symbol = pitch.substr(1, octaveStart - 1);
    const bool isValidSymbol =
        symbol.empty() ||
        std::find(c_alterSymbol.begin(), c_alterSymbol.end(), symbol) != c_alterSymbol.end();
    if (!isValidSymbol) {
        LOG_ERROR("Unknown alter symbol: " + symbol);
    }

    const float value = alterSymbol2Value(symbol);
    if (spellingToMidiNumber(step, value, parsedOctave) < 0) {
        LOG_ERROR("The pitch '" + pitch + "' is below MIDI note 0");
    }

    pitchClass = step + symbol;
    pitchStep = step;
    octave = parsedOctave;
    alterValue = value;
    alterSymbol = symbol;
}

std::string Note::getEnharmonicPitch(const bool alternativeEnhamonicPitch) const {
    if (isNoteOff()) {
        return MUSIC_XML::PITCH::REST;
    }

    const std::string pitch = getPitch();
    std::string pitchClass;
    std::string pitchStep;
    std::string alterSymbol;
    int octave = 0;
    float alterValue = 0.0f;
    Helper::splitPitch(pitch, pitchClass, pitchStep, octave, alterValue, alterSymbol);

    const int midiNumber = Helper::pitch2midiNote(pitch);
    const int ownAlter = static_cast<int>(alterValue);

    // Other spellings of the same MIDI number, indexed by 'alter + 2' (empty if unavailable)
    std::array<std::string, 5> spellings;
    for (int alter = -2; alter <= 2; alter++) {
        if (alter != ownAlter) {
            spellings[static_cast<size_t>(alter + 2)] = spellMidiNumber(midiNumber, alter);
        }
    }
    const auto spelling = [&spellings](const int alter) {
        return spellings[static_cast<size_t>(alter + 2)];
    };
    const auto firstSpelling = [&spelling](const int alterA, const int alterB) {
        return spelling(alterA).empty() ? spelling(alterB) : spelling(alterA);
    };

    std::string defaultPitch;
    std::string alternativePitch;

    const bool isWhiteKey =
        std::find(c_diatonicStepSemitones.begin(), c_diatonicStepSemitones.end(),
                  midiNumber % 12) != c_diatonicStepSemitones.end();
    if (isWhiteKey) {
        // Natural <-> flat-side spelling; the sharp-side spelling is the alternative
        const std::string flatSide = firstSpelling(-1, -2);
        const std::string sharpSide = firstSpelling(1, 2);
        if (ownAlter == 0) {
            defaultPitch = flatSide;
            alternativePitch = sharpSide;
        } else if (ownAlter < 0) {
            defaultPitch = spelling(0);
            alternativePitch = sharpSide;
        } else {
            defaultPitch = spelling(0);
            alternativePitch = flatSide;
        }
    } else {
        // Sharp <-> flat; double accidentals return the single accidental in the same direction
        const std::string doubleAccidental = firstSpelling(-2, 2);
        switch (ownAlter) {
            case 1:
                defaultPitch = spelling(-1);
                alternativePitch = doubleAccidental;
                break;
            case -1:
                defaultPitch = spelling(1);
                alternativePitch = doubleAccidental;
                break;
            case 2:
                defaultPitch = spelling(1);
                alternativePitch = spelling(-1);
                break;
            default:  // -2
                defaultPitch = spelling(-1);
                alternativePitch = spelling(1);
                break;
        }
    }

    // Range fallback: a missing default returns the own pitch; a missing alternative the default
    if (defaultPitch.empty()) {
        defaultPitch = pitch;
    }
    if (alternativePitch.empty()) {
        alternativePitch = defaultPitch;
    }

    return alternativeEnhamonicPitch ? alternativePitch : defaultPitch;
}
